using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using ProductCatalogApp.Api.Data;
using ProductCatalogApp.Api.Dtos;
using ProductCatalogApp.Api.Models;

namespace ProductCatalogApp.Api.Controllers
{
    [ApiController]
    [Route("api/[controller]")]
    [Produces("application/json")]
    public class EquipmentsController : ControllerBase
    {
        private readonly ApplicationDbContext _context;

        public EquipmentsController(ApplicationDbContext context)
        {
            _context = context;
        }

        [HttpGet]
        [ProducesResponseType(typeof(IEnumerable<EquipmentDto>), StatusCodes.Status200OK)]
        public async Task<ActionResult<IEnumerable<EquipmentDto>>> GetAll(
            [FromQuery] int? productionSiteId,
            [FromQuery] string? search)
        {
            var query = _context.EquipmentItems.AsNoTracking();

            if (productionSiteId.HasValue)
            {
                query = query.Where(e => e.ProductionSiteId == productionSiteId.Value);
            }

            if (!string.IsNullOrWhiteSpace(search))
            {
                var pattern = $"%{search.Trim()}%";
                query = query.Where(e =>
                    EF.Functions.Like(e.Name, pattern) ||
                    EF.Functions.Like(e.Model, pattern) ||
                    EF.Functions.Like(e.InventoryNumber, pattern));
            }

            var equipment = await query
                .OrderBy(e => e.ProductionSite!.Name)
                .ThenBy(e => e.Name)
                .Select(e => new EquipmentDto(
                    e.Id,
                    e.Name,
                    e.Model,
                    e.InventoryNumber,
                    e.ProductionSiteId,
                    e.ProductionSite!.Name))
                .ToListAsync();

            return Ok(equipment);
        }

        [HttpGet("{id:int}")]
        [ProducesResponseType(typeof(EquipmentDto), StatusCodes.Status200OK)]
        [ProducesResponseType(StatusCodes.Status404NotFound)]
        public async Task<ActionResult<EquipmentDto>> GetById(int id)
        {
            var equipment = await _context.EquipmentItems
                .AsNoTracking()
                .Where(e => e.Id == id)
                .Select(e => new EquipmentDto(
                    e.Id,
                    e.Name,
                    e.Model,
                    e.InventoryNumber,
                    e.ProductionSiteId,
                    e.ProductionSite!.Name))
                .FirstOrDefaultAsync();

            if (equipment == null)
            {
                return NotFound();
            }

            return Ok(equipment);
        }

        [HttpPost]
        [ProducesResponseType(typeof(EquipmentDto), StatusCodes.Status201Created)]
        [ProducesResponseType(typeof(ValidationProblemDetails), StatusCodes.Status400BadRequest)]
        public async Task<ActionResult<EquipmentDto>> Create(EquipmentInput input)
        {
            var site = await _context.ProductionSites.FindAsync(input.ProductionSiteId);

            if (site == null)
            {
                return SiteNotFound();
            }

            var equipment = new Equipment
            {
                Name = input.Name.Trim(),
                Model = input.Model.Trim(),
                InventoryNumber = input.InventoryNumber.Trim(),
                ProductionSiteId = site.Id
            };

            _context.EquipmentItems.Add(equipment);
            await _context.SaveChangesAsync();

            var dto = new EquipmentDto(
                equipment.Id,
                equipment.Name,
                equipment.Model,
                equipment.InventoryNumber,
                site.Id,
                site.Name);

            return CreatedAtAction(nameof(GetById), new { id = equipment.Id }, dto);
        }

        [HttpPut("{id:int}")]
        [ProducesResponseType(StatusCodes.Status204NoContent)]
        [ProducesResponseType(typeof(ValidationProblemDetails), StatusCodes.Status400BadRequest)]
        [ProducesResponseType(StatusCodes.Status404NotFound)]
        public async Task<IActionResult> Update(int id, EquipmentInput input)
        {
            var equipment = await _context.EquipmentItems.FindAsync(id);

            if (equipment == null)
            {
                return NotFound();
            }

            if (!await _context.ProductionSites.AnyAsync(s => s.Id == input.ProductionSiteId))
            {
                return SiteNotFound();
            }

            equipment.Name = input.Name.Trim();
            equipment.Model = input.Model.Trim();
            equipment.InventoryNumber = input.InventoryNumber.Trim();
            equipment.ProductionSiteId = input.ProductionSiteId;

            await _context.SaveChangesAsync();

            return NoContent();
        }

        [HttpDelete("{id:int}")]
        [ProducesResponseType(StatusCodes.Status204NoContent)]
        [ProducesResponseType(StatusCodes.Status404NotFound)]
        public async Task<IActionResult> Delete(int id)
        {
            var equipment = await _context.EquipmentItems.FindAsync(id);

            if (equipment == null)
            {
                return NotFound();
            }

            _context.EquipmentItems.Remove(equipment);
            await _context.SaveChangesAsync();

            return NoContent();
        }

        private ActionResult SiteNotFound()
        {
            ModelState.AddModelError(
                nameof(EquipmentInput.ProductionSiteId),
                "Производственный участок с таким идентификатором не найден.");

            return ValidationProblem(ModelState);
        }
    }
}

using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using ProductCatalogApp.Api.Data;
using ProductCatalogApp.Api.Dtos;
using ProductCatalogApp.Api.Models;

namespace ProductCatalogApp.Api.Controllers
{
    [ApiController]
    [Route("api/[controller]")]
    [Produces("application/json")]
    public class ProductionSitesController : ControllerBase
    {
        private readonly ApplicationDbContext _context;

        public ProductionSitesController(ApplicationDbContext context)
        {
            _context = context;
        }

        [HttpGet]
        [ProducesResponseType(typeof(IEnumerable<ProductionSiteDto>), StatusCodes.Status200OK)]
        public async Task<ActionResult<IEnumerable<ProductionSiteDto>>> GetAll([FromQuery] string? search)
        {
            var query = _context.ProductionSites.AsNoTracking();

            if (!string.IsNullOrWhiteSpace(search))
            {
                var pattern = $"%{search.Trim()}%";
                query = query.Where(site => EF.Functions.Like(site.Name, pattern));
            }

            var sites = await query
                .OrderBy(site => site.Name)
                .Select(site => new ProductionSiteDto(
                    site.Id,
                    site.Name,
                    site.Description,
                    site.EquipmentItems.Count))
                .ToListAsync();

            return Ok(sites);
        }

        [HttpGet("{id:int}")]
        [ProducesResponseType(typeof(ProductionSiteDetailsDto), StatusCodes.Status200OK)]
        [ProducesResponseType(StatusCodes.Status404NotFound)]
        public async Task<ActionResult<ProductionSiteDetailsDto>> GetById(int id)
        {
            var site = await _context.ProductionSites
                .AsNoTracking()
                .Include(s => s.EquipmentItems)
                .FirstOrDefaultAsync(s => s.Id == id);

            if (site == null)
            {
                return NotFound();
            }

            var equipment = site.EquipmentItems
                .OrderBy(e => e.Name)
                .Select(e => new EquipmentDto(
                    e.Id,
                    e.Name,
                    e.Model,
                    e.InventoryNumber,
                    e.ProductionSiteId,
                    site.Name))
                .ToList();

            return Ok(new ProductionSiteDetailsDto(site.Id, site.Name, site.Description, equipment));
        }

        [HttpPost]
        [ProducesResponseType(typeof(ProductionSiteDto), StatusCodes.Status201Created)]
        [ProducesResponseType(typeof(ValidationProblemDetails), StatusCodes.Status400BadRequest)]
        public async Task<ActionResult<ProductionSiteDto>> Create(ProductionSiteInput input)
        {
            var site = new ProductionSite
            {
                Name = input.Name.Trim(),
                Description = input.Description?.Trim() ?? string.Empty
            };

            _context.ProductionSites.Add(site);
            await _context.SaveChangesAsync();

            var dto = new ProductionSiteDto(site.Id, site.Name, site.Description, 0);

            return CreatedAtAction(nameof(GetById), new { id = site.Id }, dto);
        }

        [HttpPut("{id:int}")]
        [ProducesResponseType(StatusCodes.Status204NoContent)]
        [ProducesResponseType(typeof(ValidationProblemDetails), StatusCodes.Status400BadRequest)]
        [ProducesResponseType(StatusCodes.Status404NotFound)]
        public async Task<IActionResult> Update(int id, ProductionSiteInput input)
        {
            var site = await _context.ProductionSites.FindAsync(id);

            if (site == null)
            {
                return NotFound();
            }

            site.Name = input.Name.Trim();
            site.Description = input.Description?.Trim() ?? string.Empty;

            await _context.SaveChangesAsync();

            return NoContent();
        }

        [HttpDelete("{id:int}")]
        [ProducesResponseType(StatusCodes.Status204NoContent)]
        [ProducesResponseType(StatusCodes.Status404NotFound)]
        public async Task<IActionResult> Delete(int id)
        {
            var site = await _context.ProductionSites.FindAsync(id);

            if (site == null)
            {
                return NotFound();
            }

            _context.ProductionSites.Remove(site);
            await _context.SaveChangesAsync();

            return NoContent();
        }
    }
}

using Microsoft.EntityFrameworkCore;
using Microsoft.OpenApi;
using ProductCatalogApp.Api.Data;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddControllers();

builder.Services.AddDbContext<ApplicationDbContext>(options =>
    options.UseSqlite(
        builder.Configuration.GetConnectionString("DefaultConnection")));

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen(options =>
{
    options.SwaggerDoc("v1", new OpenApiInfo
    {
        Title = "Product Catalog API",
        Version = "v1"
    });
});

var app = builder.Build();

using (var scope = app.Services.CreateScope())
{
    var context = scope.ServiceProvider.GetRequiredService<ApplicationDbContext>();
    context.Database.Migrate();
    DbSeeder.Seed(context);
}

if (!app.Environment.IsDevelopment())
{

                return NotFound();

            return View(equipment);
        }

        [ValidateAntiForgeryToken]
        public async Task<IActionResult> Edit(int? id, Equipment equipment)
        {
            if (id != equipment.Id)
            {
                return NotFound();

                var result = await _api.UpdateEquipmentAsync(equipment.Id, equipment);

                    return NotFound();

using Microsoft.AspNetCore.Mvc.ModelBinding;

namespace ProductCatalogApp.Client.Services
{
    public class ApiResult
    {
        public bool Succeeded { get; private init; }

        public bool NotFound { get; private init; }

        public IReadOnlyDictionary<string, string[]> Errors { get; private init; } =
            new Dictionary<string, string[]>();

        public static ApiResult Success() => new() { Succeeded = true };

        public static ApiResult NotFoundResult() => new() { NotFound = true };

        public static ApiResult Invalid(IDictionary<string, string[]> errors) =>
            new() { Errors = new Dictionary<string, string[]>(errors) };

        public void CopyErrorsTo(ModelStateDictionary modelState)
        {
            if (NotFound)
            {
                modelState.AddModelError(string.Empty, "Запись не найдена: возможно, её уже удалили.");
                return;
            }

            if (Errors.Count == 0)
            {
                modelState.AddModelError(string.Empty, "Не удалось выполнить операцию.");
                return;
            }

            foreach (var (field, messages) in Errors)
            {
                foreach (var message in messages)
                {
                    modelState.AddModelError(field, message);
                }
            }
        }
    }
}

using System.Net;
using Microsoft.AspNetCore.Mvc;
using ProductCatalogApp.Client.Models;

namespace ProductCatalogApp.Client.Services
{
    public class ProductCatalogApiClient : IProductCatalogApiClient
    {
        private const string SitesUrl = "api/ProductionSites";
        private const string EquipmentUrl = "api/Equipments";

        private readonly HttpClient _http;

        public ProductCatalogApiClient(HttpClient http)
        {
            _http = http;
        }

        public async Task<List<ProductionSite>> GetProductionSitesAsync(string? search = null)
        {
            var url = AddQuery(SitesUrl, ("search", search));

            return await _http.GetFromJsonAsync<List<ProductionSite>>(url) ?? new();
        }

        public async Task<ProductionSiteDetails?> GetProductionSiteAsync(int id)
        {
            var response = await _http.GetAsync($"{SitesUrl}/{id}");

            if (response.StatusCode == HttpStatusCode.NotFound)
            {
                return null;
            }

            response.EnsureSuccessStatusCode();

            return await response.Content.ReadFromJsonAsync<ProductionSiteDetails>();
        }

        public async Task<ApiResult> CreateProductionSiteAsync(ProductionSite site)
        {
            return await ToResultAsync(await _http.PostAsJsonAsync(SitesUrl, site));
        }

        public async Task<ApiResult> UpdateProductionSiteAsync(int id, ProductionSite site)
        {
            return await ToResultAsync(await _http.PutAsJsonAsync($"{SitesUrl}/{id}", site));
        }

        public async Task<ApiResult> DeleteProductionSiteAsync(int id)
        {
            return await ToResultAsync(await _http.DeleteAsync($"{SitesUrl}/{id}"));
        }

        public async Task<List<Equipment>> GetEquipmentAsync(int? productionSiteId = null, string? search = null)
        {
            var url = AddQuery(
                EquipmentUrl,
                ("productionSiteId", productionSiteId?.ToString()),
                ("search", search));

            return await _http.GetFromJsonAsync<List<Equipment>>(url) ?? new();
        }

        public async Task<Equipment?> GetEquipmentByIdAsync(int id)
        {
            var response = await _http.GetAsync($"{EquipmentUrl}/{id}");

            if (response.StatusCode == HttpStatusCode.NotFound)
            {
                return null;
            }

            response.EnsureSuccessStatusCode();

            return await response.Content.ReadFromJsonAsync<Equipment>();
        }

        public async Task<ApiResult> CreateEquipmentAsync(Equipment equipment)
        {
            return await ToResultAsync(await _http.PostAsJsonAsync(EquipmentUrl, equipment));
        }

        public async Task<ApiResult> UpdateEquipmentAsync(int id, Equipment equipment)
        {
            return await ToResultAsync(await _http.PutAsJsonAsync($"{EquipmentUrl}/{id}", equipment));
        }

        public async Task<ApiResult> DeleteEquipmentAsync(int id)
        {
            return await ToResultAsync(await _http.DeleteAsync($"{EquipmentUrl}/{id}"));
        }

        private static string AddQuery(string url, params (string Name, string? Value)[] parameters)
        {
            var pairs = parameters
                .Where(p => !string.IsNullOrWhiteSpace(p.Value))
                .Select(p => $"{p.Name}={Uri.EscapeDataString(p.Value!.Trim())}")
                .ToList();

            return pairs.Count == 0 ? url : $"{url}?{string.Join('&', pairs)}";
        }

        private static async Task<ApiResult> ToResultAsync(HttpResponseMessage response)
        {
            if (response.IsSuccessStatusCode)
            {
                return ApiResult.Success();
            }

            if (response.StatusCode == HttpStatusCode.NotFound)
            {
                return ApiResult.NotFoundResult();
            }

            if (response.StatusCode == HttpStatusCode.BadRequest)
            {
                var problem = await response.Content.ReadFromJsonAsync<ValidationProblemDetails>();

                return ApiResult.Invalid(problem?.Errors ?? new Dictionary<string, string[]>());
            }

            response.EnsureSuccessStatusCode();

            return ApiResult.Success();
        }
    }
}

# Список производственных участков и оборудования (Web API)

Практическая работа №2 «Разработка приложения Web API». За основу взят проект из практической работы №1.

## Постановка задачи

Нужно создать решение ASP.NET Web API из двух проектов: серверной и клиентской части.
Тема: список производственных участков и перечень установленного на них оборудования.
В модели данных две связанные сущности (участок и оборудование, связь «один ко многим»).
В API два контроллера, в каждом есть выборка всех элементов, выборка по условию,
добавление, редактирование и удаление. API описано в Swagger, данные хранятся в SQLite.
Клиентская часть работает с данными через разработанное API.

## Что сделано

- проект `ProductCatalogApp.Api`: сущности, `DbContext`, миграция, DTO;
- контроллеры `ProductionSitesController` и `EquipmentsController`;
- фильтрация: участки ищутся по названию (`search`), оборудование фильтруется по участку (`productionSiteId`) и по тексту (`search`);
- подключён Swagger, при первом запуске база заполняется тестовыми данными;
- проект из первой работы переименован в `ProductCatalogApp.Client` и теперь берёт все данные из API.

## Запуск

Открыть `ProductCatalogApp.slnx` в Visual Studio, в свойствах решения выбрать
несколько запускаемых проектов (Api и Client) и запустить. Или запустить `run-all.bat`.

Swagger: http://localhost:5100/swagger, клиент: http://localhost:5034.
Адрес API для клиента указан в `ProductCatalogApp.Client/appsettings.json`.
